rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn(){ return request.auth != null; }
    function linkExists(){ return signedIn() && exists(/databases/$(database)/documents/deviceLinks/$(request.auth.uid)); }
    function myProfile(){ return get(/databases/$(database)/documents/deviceLinks/$(request.auth.uid)).data.profileId; }
    function isLinked(){ return linkExists(); }
    function isAdmin(){ return isLinked() && myProfile() == 'igor'; }
    function validInvite(token,pid){ return exists(/databases/$(database)/documents/invitations/$(token)) && get(/databases/$(database)/documents/invitations/$(token)).data.active == true && get(/databases/$(database)/documents/invitations/$(token)).data.profileId == pid; }

    match /invitations/{token}{ allow get: if signedIn(); allow list,create,update,delete: if isAdmin(); }
    match /deviceLinks/{uid}{
      allow get: if signedIn() && request.auth.uid == uid; allow list: if isAdmin();
      allow create: if signedIn() && request.auth.uid == uid && request.resource.data.profileId is string && request.resource.data.inviteToken is string && validInvite(request.resource.data.inviteToken,request.resource.data.profileId);
      allow update: if false; allow delete: if signedIn() && (request.auth.uid == uid || isAdmin());
    }
    match /profiles/{pid}{ allow read: if isLinked(); allow write: if isAdmin(); }
    match /preferences/{pid}{ allow read: if isLinked(); allow create,update,delete: if isLinked() && (pid == myProfile() || isAdmin()); }
    match /pushTokens/{uid}{
      allow get: if isLinked() && (request.auth.uid == uid || isAdmin()); allow list: if isAdmin();
      allow create,update: if isLinked() && request.auth.uid == uid && request.resource.data.profileId == myProfile();
      allow delete: if isLinked() && (request.auth.uid == uid || isAdmin());
    }
    match /notificationRequests/{id}{ allow create,read: if isAdmin(); allow update,delete: if false; }
    match /config/{docId}{ allow read: if isLinked(); allow write: if isAdmin(); }
    match /legacyStatus/{docId}{ allow read: if isLinked(); allow create,update,delete: if isAdmin(); }
    match /availability/{docId}{
      allow read: if isLinked();
      allow create: if isLinked() && request.resource.data.profileId == myProfile();
      allow update: if isLinked() && resource.data.profileId == myProfile() && request.resource.data.profileId == resource.data.profileId;
      allow delete: if isLinked() && (resource.data.profileId == myProfile() || isAdmin());
    }
    match /compatibilities/{docId}{
      allow read: if isLinked();
      allow create: if isLinked() && request.resource.data.responderId == myProfile();
      allow update: if isLinked() && resource.data.responderId == myProfile() && request.resource.data.responderId == resource.data.responderId;
      allow delete: if isLinked() && (resource.data.responderId == myProfile() || isAdmin());
    }
    match /plans/{date}{ allow read,create,update,delete: if isLinked(); }
    match /tripDays/{date}{ allow read,create,update,delete: if isLinked(); }
    match /{document=**}{ allow read,write: if false; }
  }
}
